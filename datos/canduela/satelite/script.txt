// Creamos el objeto Geometría
var aoi = ee.Geometry.Polygon(coordenadas);

// Fechas (puedes ajustarlas)
var FECHA_INICIO = '2023-01-01'; 
var FECHA_FIN = '2026-02-16';

// Visualización en el mapa
Map.centerObject(aoi, 16); 
Map.addLayer(aoi, {color: 'cyan'}, 'Nueva Parcela (AOI)');

// =================================================================================
// --- PASO 2: SELECCIÓN Y PROCESADO DE IMÁGENES SENTINEL-2 --- //
// =================================================================================

function procesarImagenS2(image) { 
  var qa = image.select('QA60'); 
  var cloudBitMask = 1 << 10; 
  var cirrusBitMask = 1 << 11; 
  var mask = qa.bitwiseAnd(cloudBitMask).eq(0).and(qa.bitwiseAnd(cirrusBitMask).eq(0));

  return image.updateMask(mask).divide(10000).copyProperties(image, ['system:time_start']); 
}

var coleccion_s2 = ee.ImageCollection('COPERNICUS/S2_SR_HARMONIZED') 
  .filterBounds(aoi)
  .filterDate(FECHA_INICIO, FECHA_FIN)
  .filter(ee.Filter.lt('CLOUDY_PIXEL_PERCENTAGE', 20)) 
  .map(procesarImagenS2) 
  .select(['B2', 'B3', 'B4', 'B5', 'B6', 'B7', 'B8']);

print("Imágenes encontradas en la nueva parcela:", coleccion_s2.size());

// =================================================================================
// --- PASO 3: BUCLE DE EXPORTACIÓN A GOOGLE DRIVE --- //
// =================================================================================

var listaImagenes = coleccion_s2.toList(coleccion_s2.size()); 
var numeroDeImagenes = listaImagenes.length().getInfo();

print('Preparando', numeroDeImagenes, 'tareas de exportación...');

for (var i = 0; i < numeroDeImagenes; i++) { 
  var image = ee.Image(listaImagenes.get(i));
  var fecha = image.date().format('YYYY-MM-DD').getInfo(); 
  var nombreArchivo = 'S2_Chip_NuevaParcela_' + fecha;

  Export.image.toDrive({ 
    image: image,
    description: 'Export_' + nombreArchivo, 
    folder: 'GEE_Tesis_NuevaParcela', // Se guardará en esta nueva carpeta en tu Drive
    fileNamePrefix: nombreArchivo, 
    region: aoi, 
    scale: 10, 
    fileFormat: 'GeoTIFF', 
    formatOptions: { cloudOptimized: true } 
  }); 
}
